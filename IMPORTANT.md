<script lang="ts">
	import { BRAND } from '$lib/config/brand';

	type AccountMode = 'margin' | 'cash';
	type TradeStructure = 'long_option_0dte' | 'defined_spread' | 'margin_equity';

	let accountMode = $state<AccountMode>('margin');
	let structure = $state<TradeStructure>('long_option_0dte');
	let accountEquity = $state<number>(3500);
	let settledCash = $state<number>(3500);
	let contractOrSharePrice = $state<number>(2.4);
	let spreadWidth = $state<number>(5);
	let stopLossPct = $state<number>(25);
	let maxAccountRiskPct = $state<number>(2);
	let maintenancePct = $state<number>(25); // Rule 4210 standard 25% equity maintenance (or higher house rate)
	let plannedQuantity = $state<number>(2);
	let copiedCitation = $state<boolean>(false);

	// Multiplier: 100 for standard US equity/SPX/XSP options, 1 for shares
	const unitMultiplier = $derived(structure === 'margin_equity' ? 1 : 100);

	// Capital / Maintenance requirement per unit under FINRA Rule 4210
	const costPerUnit = $derived.by(() => {
		if (structure === 'long_option_0dte') {
			// Long options & 0DTE non-marginable contracts require 100% premium commitment intraday
			return contractOrSharePrice * 100;
		}
		if (structure === 'defined_spread') {
			// Credit/debit vertical max loss width (in dollars per spread) minus/based on net risk
			// For conservative maintenance, use max loss per spread: width * 100 - credit (or debit paid)
			return Math.max(contractOrSharePrice * 100, (spreadWidth - contractOrSharePrice) * 100);
		}
		return contractOrSharePrice;
	});

	const totalNotionalExposure = $derived(
		structure === 'margin_equity'
			? contractOrSharePrice * plannedQuantity
			: costPerUnit * plannedQuantity
	);

	const intradayMaintenanceRequired = $derived.by(() => {
		if (accountMode === 'cash') {
			return totalNotionalExposure;
		}
		if (structure === 'margin_equity') {
			return totalNotionalExposure * (maintenancePct / 100);
		}
		// Options premium / defined-spread max risk carries 100% maintenance requirement
		return totalNotionalExposure;
	});

	// FINRA Rule 4210(d)(2): Intraday Margin Level (IML) = Account Equity - Maintenance Requirement
	const intradayMarginLevel = $derived(accountEquity - intradayMaintenanceRequired);
	const intradayMarginDeficit = $derived(Math.max(0, -intradayMarginLevel));

	// FINRA Notice 26-10 De Minimis Exception: lesser of 5% of account equity or $1,000
	const deMinimisLimit = $derived(Math.min(accountEquity * 0.05, 1000));

	const marginStatus = $derived.by(() => {
		if (accountMode === 'cash') {
			return totalNotionalExposure <= settledCash
				? {
						code: 'SAFE',
						label: 'T+1 Settled Cash Compliant',
						detail:
							'Position fits strictly inside settled cash—zero Good Faith Violation (GFV) or margin freeze risk.'
					}
				: {
						code: 'BLOCK',
						label: 'Exceeds Settled Cash (GFV Risk)',
						detail:
							'Closing before settlement triggers a Good Faith Violation. Reduce contracts or wait for T+1 options settlement.'
					};
		}

		if (accountEquity < 2000) {
			return {
				code: 'BLOCK',
				label: 'Below $2,000 FINRA Margin Floor',
				detail:
					'While the $25,000 PDT rule ended June 4, 2026, FINRA Rule 4210 retains the $2,000 minimum equity floor for leveraged margin accounts. Use a T+1 Cash Account below $2,000.'
			};
		}

		if (intradayMarginDeficit === 0) {
			return {

			return contractOrSharePrice * 100;
		}

			return Math.max(contractOrSharePrice * 100, (spreadWidth - contractOrSharePrice) * 100);
		}
		return contractOrSharePrice;

import { BRAND } from '$lib/config/brand';
import type { RouteSEO } from './seo-routes';

export function buildRouteJsonLd(route: RouteSEO): string {
	const canonicalUrl = `${BRAND.domain}${route.path === '/' ? '/' : route.path}`;
	const orgId = `${BRAND.domain}/#organization`;

			logo: {
				'@type': 'ImageObject',
				url: `${BRAND.domain}/images/logo-512x512.png`,
				width: 512,
				height: 512
			},
			founder: { '@id': founderId },
			sameAs: BRAND.socials,
			knowsAbout: [
				'Price Action Trading Without Indicators',
				'Multi-Timeframe Market Structure',
				'The Move Prior to The Move',

			]
		},

			'@type': 'Person',
			'@id': founderId,
			name: BRAND.founder.name,
			alternateName: BRAND.founder.shortName,
			jobTitle: BRAND.founder.jobTitle,
			worksFor: { '@id': orgId },
			description: BRAND.founder.mentorshipLineage,
			subjectOf: BRAND.founder.corroborationUrls
		}
	];

	// Google Search Central rule: WebSite node ONLY on homepage
	if (route.path === '/') {
		graph.unshift({
			'@type': 'WebSite',
			'@id': `${BRAND.domain}/#website`,
			url: `${BRAND.domain}/`,
			name: BRAND.name,
			publisher: { '@id': orgId }
		});
	} else {
		// BreadcrumbList on all inner routes
		graph.push({
			'@type': 'BreadcrumbList',
			'@id': `${canonicalUrl}#breadcrumb`,
			itemListElement: [
				{ '@type': 'ListItem', position: 1, name: 'Home', item: `${BRAND.domain}/` },
				{ '@type': 'ListItem', position: 2, name: route.h1, item: canonicalUrl }
			]
		});
	}

	if (route.schemaType === 'SoftwareApplication') {
		graph.push({
			'@type': 'WebApplication',
			name: route.h1,
			url: canonicalUrl,
			applicationCategory: 'FinanceApplication',
			operatingSystem: 'All',
			offers: { '@type': 'Offer', price: '0', priceCurrency: 'USD' },

			'@type': 'Article',
			headline: route.h1,
			description: route.description,
			url: canonicalUrl,
			author: { '@id': founderId },
			publisher: { '@id': orgId },
			mainEntityOfPage: canonicalUrl
		});
	}

	return JSON.stringify({ '@context': 'https://schema.org', '@graph': graph });

// src/routes/sitemap.xml/+server.ts
import type { RequestHandler } from './$types';
import { BRAND } from '$lib/config/brand';
import { SEO_ROUTES } from '$lib/seo/seo-routes';

export const GET: RequestHandler = async () => {
	const today = new Date().toISOString().split('T')[0];
	const entries = Object.values(SEO_ROUTES)
		.map(
			(r) => `  <url>
    <loc>${BRAND.domain}${r.path === '/' ? '/' : r.path}</loc>
    <lastmod>${today}</lastmod>
    <changefreq>${r.changefreq}</changefreq>
    <priority>${r.priority.toFixed(1)}</priority>
  </url>`
		)

	const xml = `<?xml version="1.0" encoding="UTF-8"?>
<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">
${entries}
</urlset>`;

	return new Response(xml, {
		headers: {
			'Content-Type': 'application/xml; charset=utf-8',
			'Cache-Control': 'public, max-age=3600'
		}
	});
